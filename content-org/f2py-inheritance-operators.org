** Type inheritance via extends()

*** The Fortran types

Start small. ~Particle~ has ~mass~ and ~charge~. ~ChargedParticle~ extends it and adds ~spin~.

#+begin_src f90
module inheritance_mod
  use iso_c_binding
  implicit none

  type :: Particle
    double precision :: mass
    integer :: charge
  end type Particle

  type, extends(Particle) :: ChargedParticle
    double precision :: spin
  end type ChargedParticle

contains
  subroutine scale_mass(p, factor)
    type(Particle), intent(inout) :: p
    double precision, intent(in) :: factor
    p%mass = p%mass * factor
  end subroutine
end module
#+end_src

A ~ChargedParticle~ has ~mass~ and ~charge~ from ~Particle~, and its own ~spin~. The standard calls that type extension (F2018 7.5.7.1).

*** Manual wrappers

One might expect the child to reuse the parent's getters. It can't. Each type gets its own wrappers, and the child reads the inherited members through its own layout[fn:layoutluck]:

#+end_export

#+begin_src f90
module particle_wrappers
  use inheritance_mod
  use iso_c_binding
  implicit none
contains
  function particle_create(mass, charge) result(cptr) bind(c)
    real(c_double), value :: mass
    integer(c_int), value :: charge
    type(c_ptr) :: cptr
    type(Particle), pointer :: obj
    allocate(obj)
    obj%mass = mass; obj%charge = charge
    cptr = c_loc(obj)
  end function

  subroutine particle_destroy(cptr) bind(c)
    type(c_ptr), value :: cptr
    type(Particle), pointer :: obj
    call c_f_pointer(cptr, obj)
    deallocate(obj)
  end subroutine

  function particle_get_mass(cptr) result(val) bind(c)
    type(c_ptr), value :: cptr
    real(c_double) :: val
    type(Particle), pointer :: obj
    call c_f_pointer(cptr, obj)
    val = obj%mass
  end function

  function particle_get_charge(cptr) result(val) bind(c)
    type(c_ptr), value :: cptr
    integer(c_int) :: val
    type(Particle), pointer :: obj
    call c_f_pointer(cptr, obj)
    val = obj%charge
  end function

  ! Child: ChargedParticle gets its OWN accessors for inherited members
  function charged_create(mass, charge, spin) result(cptr) bind(c)
    real(c_double), value :: mass
    integer(c_int), value :: charge
    real(c_double), value :: spin
    type(c_ptr) :: cptr
    type(ChargedParticle), pointer :: obj
    allocate(obj)
    obj%mass = mass; obj%charge = charge; obj%spin = spin
    cptr = c_loc(obj)
  end function

  subroutine charged_destroy(cptr) bind(c)
    type(c_ptr), value :: cptr
    type(ChargedParticle), pointer :: obj
    call c_f_pointer(cptr, obj)
    deallocate(obj)
  end subroutine

  function charged_get_mass(cptr) result(val) bind(c)
    type(c_ptr), value :: cptr
    real(c_double) :: val
    type(ChargedParticle), pointer :: obj
    call c_f_pointer(cptr, obj)
    val = obj%mass   ! accesses via ChargedParticle layout
  end function

  function charged_get_spin(cptr) result(val) bind(c)
    type(c_ptr), value :: cptr
    real(c_double) :: val
    type(ChargedParticle), pointer :: obj
    call c_f_pointer(cptr, obj)
    val = obj%spin
  end function
end module
#+end_src

#+begin_export html
</details>
#+end_export

~charged_get_mass~ does ~c_f_pointer~ into ~type(ChargedParticle)~, not ~type(Particle)~, and then reads ~obj%mass~. The capsule holds a ~ChargedParticle~. Reading that memory as a ~Particle~ works on every compiler I have tried[fn:layoutluck], and the standard does not promise it, so the wrappers stay type-specific.

*** Python-side inheritance via tp_base

CPython does single inheritance with ~tp_base~. Point ~ChargedParticleType.tp_base~ at ~ParticleType~ and ~isinstance(cp, ParticleType)~ is true:

#+begin_src c
static PyTypeObject PyChargedParticle_Type = {
    PyVarObject_HEAD_INIT(NULL, 0)
    .tp_name = "charged_particle",
    .tp_basicsize = sizeof(PyChargedParticleObject),
    .tp_base = &PyParticle_Type,    /* inheritance */
    /* ... */
};
#+end_src

~tp_base~ is what makes ~isinstance()~ work. It does not hand you the attributes. Each type still has its own getset table, because the opaque pointer needs an accessor that knows the type.

** Operator overloading

*** The Fortran module

#+begin_src f90
module operator_mod
  use iso_c_binding
  implicit none

  type :: Vec2
    double precision :: x, y
  end type Vec2

  interface operator(+)
    module procedure vec2_add
  end interface
  interface operator(==)
    module procedure vec2_eq
  end interface

contains
  function vec2_add(a, b) result(c)
    type(Vec2), intent(in) :: a, b
    type(Vec2) :: c
    c%x = a%x + b%x
    c%y = a%y + b%y
  end function

  function vec2_eq(a, b) result(eq)
    type(Vec2), intent(in) :: a, b
    logical :: eq
    eq = (a%x == b%x) .and. (a%y == b%y)
  end function
end module
#+end_src

*** Manual wrappers for operators

Addition takes two ~c_ptr~ arguments and returns a new ~c_ptr~. Comparison returns a boolean. The wrapper module also needs create, destroy, and the getters. That part is the same mechanical pattern as the other posts:

<details>
<summary>Full Vec2 wrapper module (click to expand)</summary>
#+end_export

#+begin_src f90
module vec2_wrappers
  use operator_mod
  use iso_c_binding
  implicit none
contains
  function vec2_create(x, y) result(cptr) bind(c)
    real(c_double), value :: x, y
    type(c_ptr) :: cptr
    type(Vec2), pointer :: obj
    allocate(obj)
    obj%x = x; obj%y = y
    cptr = c_loc(obj)
  end function

  subroutine vec2_destroy(cptr) bind(c)
    type(c_ptr), value :: cptr
    type(Vec2), pointer :: obj
    call c_f_pointer(cptr, obj)
    deallocate(obj)
  end subroutine

  function vec2_get_x(cptr) result(val) bind(c)
    type(c_ptr), value :: cptr
    real(c_double) :: val
    type(Vec2), pointer :: obj
    call c_f_pointer(cptr, obj)
    val = obj%x
  end function

  function vec2_get_y(cptr) result(val) bind(c)
    type(c_ptr), value :: cptr
    real(c_double) :: val
    type(Vec2), pointer :: obj
    call c_f_pointer(cptr, obj)
    val = obj%y
  end function

  function vec2_wrap_add(a_ptr, b_ptr) result(c_ptr_out) bind(c)
    type(c_ptr), value :: a_ptr, b_ptr
    type(c_ptr) :: c_ptr_out
    type(Vec2), pointer :: a, b, c_obj
    call c_f_pointer(a_ptr, a)
    call c_f_pointer(b_ptr, b)
    allocate(c_obj)
    c_obj = vec2_add(a, b)
    c_ptr_out = c_loc(c_obj)
  end function

  function vec2_wrap_eq(a_ptr, b_ptr) result(eq) bind(c)
    type(c_ptr), value :: a_ptr, b_ptr
    logical(c_bool) :: eq
    type(Vec2), pointer :: a, b
    call c_f_pointer(a_ptr, a)
    call c_f_pointer(b_ptr, b)
    eq = vec2_eq(a, b)
  end function
end module
#+end_src

*** Python number protocol

CPython puts arithmetic in ~PyNumberMethods~ (~nb_add~, ~nb_subtract~) and comparison in ~tp_richcompare~. The generated C checks both operands and returns ~NotImplemented~ when either has the wrong type[fn:notimpl]:

#+begin_src c
static PyObject *
Pyvec2_nb_add(PyObject *left, PyObject *right) {
    if (!Py_IS_TYPE(left, &Pyvec2_Type) ||
        !Py_IS_TYPE(right, &Pyvec2_Type)) {
        Py_RETURN_NOTIMPLEMENTED;
    }
    /* extract capsules, call vec2_wrap_add, wrap result */
}
#+end_src

** f2py automation

~f2py~ sees ~extends()~ in what crackfortran parsed and sets ~tp_base~. It writes a full getset table for each type, and the child lists the parent's members again. Operator interfaces in the module body go to ~PyNumberMethods~ for arithmetic and ~tp_richcompare~ for comparison. I re-ran the inheritance and operator tests at the series tip, and those passed. The abstract-type tests on that same commit do not, so this post does not claim those.

** Conclusions

Type extension becomes ~tp_base~, so ~isinstance~ works. The child still lists the parent's getset entries, because each accessor has to know which type the capsule is.

Operator overloading becomes ~PyNumberMethods~ and ~tp_richcompare~. A type mismatch returns ~NotImplemented~, and that is how Python tries the reflected operator before it gives up.

[fn:f18ref] The F2018 section numbers are from the J3/18-007r1 interpretation document (WD 1539-1).
[fn:layoutluck] Inherited components come first in the layout on every compiler I have tried. That is not a promise in the standard, so the generated accessors stay type-specific.
[fn:notimpl] If ~a + b~ does not work, CPython tries ~b.__radd__(a)~ before it fails. Returning ~NotImplemented~ is what opts into that.
